### 关键实现代码

#### 1. WebSocket 消息接收循环

```python
async def run(self):
    """会话主循环"""
    try:
        # 持续接收消息
        async for message in self.websocket:
            await self.handle_message(message)
    except Exception as e:
        logger.error(f"[{self.session_id}] 会话错误: {e}", exc_info=True)

async def handle_message(self, message):
    """处理接收到的消息"""

    if isinstance(message, bytes):
        # 二进制音频数据
        await self.handle_audio(message)
    elif isinstance(message, str):
        # 文本消息（元数据、配置等）
        try:
            data = json.loads(message)
            await self.handle_text_message(data)
        except json.JSONDecodeError:
            logger.warning(f"[{self.session_id}] 无效的 JSON 消息")
```

**关键点**:
- 使用 `async for` 循环持续接收
- 通过 `isinstance(message, bytes)` 区分二进制音频和文本消息
- 二进制消息是原始 PCM 音频数据

#### 2. 音频数据处理

```python
async def handle_audio(self, audio_data: bytes):
    """处理音频数据"""

    if self.is_processing:
        # 正在处理中，忽略新音频（防止并发）
        return

    # 添加到缓冲区
    self.audio_buffer.append(audio_data)

    # VAD 检测
    if self.vad:
        speech_ended = await self._check_vad(audio_data)

        if speech_ended:
            # 检测到语音结束，立即处理
            await self._process_speech()
    else:
        # 无 VAD，检查缓冲区是否达到最小时长
        if self.audio_buffer.is_ready():
            await self._process_speech()
```

**关键点**:
- 防止并发处理（`is_processing` 标志）
- 支持两种触发模式：
  - VAD 模式：语音结束时触发
  - 定时模式：累积到最小时长触发

#### 3. VAD 处理

```python
async def _check_vad(self, audio_data: bytes) -> bool:
    """检查 VAD，返回是否语音结束"""

    # VAD 需要固定大小的帧
    frame_size = self.vad.frame_size  # 如 480 bytes (30ms @ 8kHz)

    for i in range(0, len(audio_data), frame_size):
        frame = audio_data[i:i+frame_size]

        if len(frame) == frame_size:
            # 处理每一帧
            is_speech, speech_ended = self.vad.process_frame(frame)

            if speech_ended:
                return True

    return False
```

**VAD 状态机**:
```
┌─────────┐  语音开始   ┌─────────┐  静音开始   ┌─────────┐
│  静音   │ ─────────► │  语音   │ ─────────► │  静音   │
│  状态   │            │  状态   │            │  计数   │
└─────────┘            └─────────┘            └────┬────┘
                                                    │
                                                    │ 静音超过阈值
                                                    ↓
                                              语音结束信号
```

#### 4. 完整语音处理

```python
async def _process_speech(self):
    """处理完整语音"""

    # 设置处理标志
    self.is_processing = True

    try:
        # 1. 获取音频数据
        audio_data = self.audio_buffer.get_and_clear()

        if not audio_data:
            return

        logger.info(f"[{self.session_id}] 处理音频: {len(audio_data)} bytes")

        # 2. STT: 语音识别
        text = await self.stt_adapter.transcribe(
            audio_data=audio_data,
            sample_rate=self.config.stt_sample_rate,
            language=self.config.stt_language
        )

        if not text:
            logger.warning(f"[{self.session_id}] STT 未识别到文本")
            return

        logger.info(f"[{self.session_id}] STT 结果: {text}")

        # 3. 更新对话历史
        self.conversation_history.append({
            "role": "user",
            "content": text
        })

        # 4. LLM: 生成回复
        response = await self.llm_adapter.chat(
            messages=self.conversation_history
        )

        logger.info(f"[{self.session_id}] LLM 回复: {response}")

        # 5. 更新对话历史
        self.conversation_history.append({
            "role": "assistant",
            "content": response
        })

        # 6. TTS: 文本转语音
        audio_response = await self.tts_adapter.synthesize(
            text=response,
            voice=self.config.tts_voice,
            sample_rate=self.config.tts_sample_rate
        )

        logger.info(f"[{self.session_id}] TTS 生成: {len(audio_response)} bytes")

        # 7. 发送音频（见下一章节）
        await self.send_audio(audio_response)

    except Exception as e:
        logger.error(f"[{self.session_id}] 处理语音错误: {e}", exc_info=True)
    finally:
        self.is_processing = False
```

### 两种实现方式对比

#### 方式一：HTTPS TTS（推荐，简单）

```python
async def send_to_mod_audio_stream_https(self, text: str):
    """使用 HTTPS TTS 并发送"""

    # 1. 一次性获取完整音频
    audio_data = await self.tts_adapter.synthesize(
        text=text,
        voice=self.config.tts_voice,
        sample_rate=self.config.tts_sample_rate
    )

    if not audio_data:
        logger.warning("TTS 返回空音频")
        return False

    # 2. Base64 编码
    audio_base64 = base64.b64encode(audio_data).decode('utf-8')

    # 3. 构建消息
    message = {
        "type": "streamAudio",
        "data": {
            "audioDataType": "raw",
            "sampleRate": self.config.tts_sample_rate,
            "audioData": audio_base64
        }
    }

    # 4. 发送
    await self.websocket.send(json.dumps(message))

    logger.info(f"音频已发送: {len(audio_data)} bytes")
    return True
```

**特点**:
- ✅ 简单直接，一次发送
- ✅ 代码清晰，易于维护
- ⚠️ 需要等待完整音频生成（200-800ms）
- ⚠️ 大音频占用内存

#### 方式二：WebSocket 流式 TTS（低延迟）

```python
async def send_to_mod_audio_stream_streaming(self, text: str):
    """使用 WebSocket 流式 TTS 并发送"""

    audio_buffer = bytearray()
    chunk_size = 16000  # 每 16KB 发送一次
    chunk_count = 0

    # 1. 流式接收 TTS 音频
    async for audio_chunk in self.tts_adapter.synthesize_stream(text):
        # 累积到缓冲区
        audio_buffer.extend(audio_chunk)

        # 当缓冲区达到阈值时，发送一次
        while len(audio_buffer) >= chunk_size:
            # 取出一块数据
            send_chunk = bytes(audio_buffer[:chunk_size])
            audio_buffer = audio_buffer[chunk_size:]

            # 编码并发送
            audio_base64 = base64.b64encode(send_chunk).decode('utf-8')

            message = {
                "type": "streamAudio",
                "data": {
                    "audioDataType": "raw",
                    "sampleRate": self.config.tts_sample_rate,
                    "audioData": audio_base64
                }
            }

            await self.websocket.send(json.dumps(message))
            chunk_count += 1

            logger.debug(f"发送音频块 #{chunk_count}: {len(send_chunk)} bytes")

    # 2. 发送剩余数据
    if len(audio_buffer) > 0:
        send_chunk = bytes(audio_buffer)
        audio_base64 = base64.b64encode(send_chunk).decode('utf-8')

        message = {
            "type": "streamAudio",
            "data": {
                "audioDataType": "raw",
                "sampleRate": self.config.tts_sample_rate,
                "audioData": audio_base64
            }
        }

        await self.websocket.send(json.dumps(message))
        chunk_count += 1

    logger.info(f"流式音频发送完成: 共 {chunk_count} 块")
    return True
```

**特点**:
- ✅ 首字延迟极低（< 100ms）
- ✅ 边生成边播放，用户体验好
- ✅ 内存占用低（流式处理）
- ⚠️ 实现复杂，需要缓冲管理
- ⚠️ 需要处理多次发送

## 关键代码实现

### AudioBuffer 实现

```python
class AudioBuffer:
    """音频缓冲器"""

    def __init__(self, sample_rate: int = 8000, min_duration: float = 0.5):
        self.sample_rate = sample_rate
        self.min_duration = min_duration
        self.buffer = bytearray()

    def append(self, data: bytes):
        """添加音频数据"""
        self.buffer.extend(data)

    def get_duration(self) -> float:
        """获取缓冲区音频时长（秒）"""
        # L16 PCM: 2 bytes per sample
        samples = len(self.buffer) // 2
        return samples / self.sample_rate

    def is_ready(self) -> bool:
        """是否有足够的音频可处理"""
        return self.get_duration() >= self.min_duration

    def get_and_clear(self) -> bytes:
        """获取并清空缓冲区"""
        data = bytes(self.buffer)
        self.buffer.clear()
        return data

    def clear(self):
        """清空缓冲区"""
        self.buffer.clear()
```

### VAD 处理器实现

```python
import webrtcvad

class VADProcessor:
    """语音活动检测"""

    def __init__(self, sample_rate: int = 8000, aggressiveness: int = 2):
        """
        初始化 VAD

        参数:
            sample_rate: 采样率 (8000, 16000, 32000, 48000)
            aggressiveness: 激进度 (0-3，越大越严格)
        """
        self.vad = webrtcvad.Vad(aggressiveness)
        self.sample_rate = sample_rate
        self.frame_duration = 30  # ms
        self.frame_size = int(sample_rate * self.frame_duration / 1000) * 2  # bytes

        # 状态管理
        self.is_speaking = False
        self.silence_frames = 0
        self.speech_frames = 0
        self.max_silence_frames = 20  # 600ms 静音后结束
        self.min_speech_frames = 3    # 90ms 语音后开始

    def process_frame(self, frame: bytes) -> tuple:
        """
        处理音频帧

        返回:
            (is_speech, speech_ended)
        """

        if len(frame) != self.frame_size:
            return (False, False)

        # 检测是否为语音
        try:
            is_speech = self.vad.is_speech(frame, self.sample_rate)
        except:
            return (False, False)

        if is_speech:
            self.speech_frames += 1
            self.silence_frames = 0

            if not self.is_speaking and self.speech_frames >= self.min_speech_frames:
                # 开始说话
                self.is_speaking = True
        else:
            self.silence_frames += 1

            if self.is_speaking and self.silence_frames >= self.max_silence_frames:
                # 结束说话
                self.is_speaking = False
                self.speech_frames = 0
                return (False, True)  # 语音结束

        return (is_speech, False)

    def reset(self):
        """重置状态"""
        self.is_speaking = False
        self.silence_frames = 0
        self.speech_frames = 0
```

### 示例 1: 完整的 SessionHandler

```python
import asyncio
import json
import base64
import logging
from websockets.server import WebSocketServerProtocol

logger = logging.getLogger(__name__)


class SessionHandler:
    """会话处理器 - 完整实现"""

    def __init__(self, websocket: WebSocketServerProtocol,
                 session_id: str, config):
        self.websocket = websocket
        self.session_id = session_id
        self.config = config

        # 初始化组件
        self.audio_buffer = AudioBuffer(
            sample_rate=config.stt_sample_rate,
            min_duration=0.5
        )

        self.vad = VADProcessor(
            sample_rate=config.stt_sample_rate,
            aggressiveness=2
        )

        # 初始化服务适配器
        self.stt_adapter = QwenSTTAdapter(api_key=config.stt_api_key)
        self.llm_adapter = QwenLLMAdapter(api_key=config.llm_api_key)
        self.tts_adapter = QwenHTTPSTTSAdapter(api_key=config.tts_api_key)

        # 对话历史
        self.conversation_history = []

        # 处理标志
        self.is_processing = False

    async def run(self):
        """主循环"""
        logger.info(f"[{self.session_id}] 会话开始")

        try:
            async for message in self.websocket:
                await self.handle_message(message)
        except Exception as e:
            logger.error(f"[{self.session_id}] 错误: {e}", exc_info=True)
        finally:
            await self.cleanup()

    async def handle_message(self, message):
        """消息分发"""
        if isinstance(message, bytes):
            await self.handle_audio(message)
        elif isinstance(message, str):
            try:
                data = json.loads(message)
                await self.handle_text_message(data)
            except json.JSONDecodeError:
                logger.warning(f"[{self.session_id}] 无效 JSON")

    async def handle_audio(self, audio_data: bytes):
        """处理音频"""
        if self.is_processing:
            return

        # 缓冲
        self.audio_buffer.append(audio_data)

        # VAD 检测
        speech_ended = await self._check_vad(audio_data)

        if speech_ended:
            await self._process_speech()

    async def _check_vad(self, audio_data: bytes) -> bool:
        """VAD 检测"""
        frame_size = self.vad.frame_size

        for i in range(0, len(audio_data), frame_size):
            frame = audio_data[i:i+frame_size]
            if len(frame) == frame_size:
                _, speech_ended = self.vad.process_frame(frame)
                if speech_ended:
                    return True
        return False

    async def _process_speech(self):
        """处理完整语音"""
        self.is_processing = True

        try:
            # 1. 获取音频
            audio_data = self.audio_buffer.get_and_clear()
            if not audio_data:
                return

            logger.info(f"[{self.session_id}] 处理 {len(audio_data)} bytes")

            # 2. STT
            text = await self.stt_adapter.transcribe(
                audio_data=audio_data,
                sample_rate=self.config.stt_sample_rate,
                language="zh-CN"
            )

            if not text:
                return

            logger.info(f"[{self.session_id}] 识别: {text}")

            # 3. 对话历史
            self.conversation_history.append({
                "role": "user",
                "content": text
            })

            # 4. LLM
            response = await self.llm_adapter.chat(
                messages=self.conversation_history
            )

            logger.info(f"[{self.session_id}] 回复: {response}")

            self.conversation_history.append({
                "role": "assistant",
                "content": response
            })

            # 5. TTS
            audio_response = await self.tts_adapter.synthesize(
                text=response,
                voice=self.config.tts_voice,
                sample_rate=self.config.tts_sample_rate
            )

            logger.info(f"[{self.session_id}] TTS: {len(audio_response)} bytes")

            # 6. 发送
            await self.send_audio(audio_response)

        finally:
            self.is_processing = False

    async def send_audio(self, audio_data: bytes):
        """发送音频到 mod_audio_stream"""

        # Base64 编码
        audio_base64 = base64.b64encode(audio_data).decode('utf-8')

        # 构建消息
        message = {
            "type": "streamAudio",
            "data": {
                "audioDataType": "raw",
                "sampleRate": self.config.tts_sample_rate,
                "audioData": audio_base64
            }
        }

        # 发送
        await self.websocket.send(json.dumps(message))

        logger.info(f"[{self.session_id}] 音频已发送")

    async def handle_text_message(self, data: dict):
        """处理文本消息"""
        msg_type = data.get("type")

        if msg_type == "config":
            # 处理配置更新
            logger.info(f"[{self.session_id}] 配置更新: {data}")

    async def cleanup(self):
        """清理资源"""
        logger.info(f"[{self.session_id}] 清理")

        if hasattr(self.stt_adapter, 'close'):
            await self.stt_adapter.close()
        if hasattr(self.llm_adapter, 'close'):
            await self.llm_adapter.close()
        if hasattr(self.tts_adapter, 'close'):
            await self.tts_adapter.close()
```

### 示例 2: WebSocket 服务器

```python
import asyncio
import websockets
from websockets.server import WebSocketServerProtocol


class AudioStreamServer:
    """WebSocket 音频流服务器"""

    def __init__(self, host: str, port: int, config):
        self.host = host
        self.port = port
        self.config = config
        self.sessions = {}

    async def start(self):
        """启动服务器"""
        logger.info(f"启动服务器: ws://{self.host}:{self.port}")

        async with websockets.serve(
            self.handle_connection,
            self.host,
            self.port,
            ping_interval=20,
            ping_timeout=10
        ):
            await asyncio.Future()  # 永久运行

    async def handle_connection(self,
                               websocket: WebSocketServerProtocol,
                               path: str):
        """处理新连接"""
        remote_addr = websocket.remote_address
        session_id = f"{remote_addr[0]}_{remote_addr[1]}_{id(websocket)}"

        logger.info(f"新连接: {session_id} from {remote_addr}")

        try:
            # 创建会话处理器
            handler = SessionHandler(
                websocket=websocket,
                session_id=session_id,
                config=self.config
            )

            self.sessions[session_id] = handler

            # 运行会话
            await handler.run()

        except websockets.exceptions.ConnectionClosed:
            logger.info(f"连接关闭: {session_id}")
        except Exception as e:
            logger.error(f"连接错误: {e}", exc_info=True)
        finally:
            # 移除会话
            if session_id in self.sessions:
                del self.sessions[session_id]

            logger.info(f"会话结束: {session_id}")


# 启动服务
async def main():
    config = ServiceConfig.from_env()
    server = AudioStreamServer(
        host="0.0.0.0",
        port=8080,
        config=config
    )
    await server.start()


if __name__ == "__main__":
    asyncio.run(main())
```

## 总结

### 接收音频流的关键点

1. **使用二进制帧**: mod_audio_stream 发送原始 PCM 二进制数据
2. **持续接收**: `async for message in websocket`
3. **类型判断**: `isinstance(message, bytes)` 识别音频
4. **缓冲管理**: 累积音频到足够长度
5. **VAD 检测**: 智能判断语音结束点
6. **异步处理**: STT → LLM → TTS 流程

### 返回音频流的关键点

1. **获取音频**: TTS API 返回 PCM bytes
2. **Base64 编码**: 二进制转文本
3. **JSON 封装**: `{"type": "streamAudio", "data": {...}}`
4. **文本帧发送**: `websocket.send(json.dumps(message))`
5. **格式匹配**: audioDataType 和 sampleRate 必须正确

### 最佳实践

1. ✅ 使用 VAD 提高用户体验
2. ✅ 异步非阻塞处理
3. ✅ 完善的错误处理和日志
4. ✅ 会话隔离（避免并发冲突）
5. ✅ 资源清理（连接关闭时）
6. ✅ 默认使用 HTTPS TTS（简单可靠）
7. ✅ 需要低延迟时使用 WebSocket TTS
